import json

from typing import List, Optional

from requests.adapters import HTTPAdapter

DEFAULT_QWEN_MODEL = "qwen-plus"
DEFAULT_QWEN_TEMPERATURE = 0.2
DEFAULT_QWEN_MAX_TOKENS = 2000

def _read_repo_list() -> List[str]:
    """Return the configured target repositories as a list of repository identifiers."""
    raw = os.getenv("REPOSITORIES") or os.getenv("TARGET_REPOSITORIES") or ""
    repos = []
    for item in raw.split(","):
        cleaned = item.strip()
        if cleaned:
            repos.append(cleaned)
    return repos


def _normalize_repo_names(repos: Optional[List[str]] = None) -> List[str]:
    """Normalize repo names to full names: user/repo or repo if already qualified."""
    target_repos = repos if repos is not None else _read_repo_list()
    github_user = (os.getenv("GITHUB_USER") or DEFAULT_GITHUB_USER).strip()
    normalized: List[str] = []

    for repo in target_repos:
        repo_name = repo.strip()
        if not repo_name:
            continue
        if "/" in repo_name:
            normalized.append(repo_name)
        else:
            normalized.append(f"{github_user}/{repo_name}")

    return normalized


def _build_session_with_retry() -> requests.Session:
    """Build a requests.Session with exponential backoff retry strategy (F4)."""
    session = requests.Session()
    retry_strategy = Retry(
        total=3,
        backoff_factor=1.5,
        status_forcelist=[429, 500, 502, 503, 504],
        allowed_methods=["POST"],
    )
    adapter = HTTPAdapter(max_retries=retry_strategy)
    session.mount("https://", adapter)
    session.mount("http://", adapter)
    return session


class QwenClient:
    """Small OpenAI-compatible client for the Alibaba Qwen API."""

    def __init__(
        self,
        api_key: Optional[str] = None,
        api_url: Optional[str] = None,
        model: Optional[str] = None,
        temperature: Optional[float] = None,
        max_tokens: Optional[int] = None,
        require_key: bool = True,
    ):

        self.api_url = (api_url or os.getenv("QWEN_API_URL") or DEFAULT_QWEN_API_URL).rstrip("/")
        self.model = (model or os.getenv("QWEN_MODEL") or DEFAULT_QWEN_MODEL).strip()

            )
        except ValueError:
            self.temperature = DEFAULT_QWEN_TEMPERATURE
        
        try:
            self.max_tokens = int(
                os.getenv("QWEN_MAX_TOKENS") or str(max_tokens or DEFAULT_QWEN_MAX_TOKENS)
            )
        except ValueError:
            self.max_tokens = DEFAULT_QWEN_MAX_TOKENS

        if require_key and not self.api_key:
            raise ValueError("QWEN_API_KEY is missing. Set it in the environment or .env file.")

            "Ta priorité est la sécurité, la clarté, la fiabilité et la conformité aux bonnes pratiques. "

            "Tu dois répondre en français, de façon concise et utile, avec une structure claire. "
            f"Ajoute toujours le marqueur {self.marker} à la fin de chaque réponse."
        )

    def ask(self, prompt: str) -> str:
        """Send a prompt to Qwen and return the textual answer."""
        if not prompt or not prompt.strip():
            raise ValueError("The prompt cannot be empty.")

        payload = {
            "model": self.model,
            "messages": [
                {"role": "system", "content": self._build_system_prompt()},
                {"role": "user", "content": prompt.strip()},
            ],
            "temperature": self.temperature,
            "max_tokens": self.max_tokens,
        }

            "Authorization": f"Bearer {self.api_key}",
            "Content-Type": "application/json",
        }

                headers=headers,
                data=json.dumps(payload),

                    error_body = {"error": response.text}

                    error_body.get("error", {}).get("message")
                    if isinstance(error_body.get("error"), dict)
                    else error_body
                )
                raise RuntimeError(f"Qwen API error ({response.status_code}): {message}")

            try:
                data = response.json()
            except ValueError as exc:
                raise RuntimeError("Invalid JSON returned by the Qwen API.") from exc

            choices = data.get("choices") or []
            if not choices:
                raise RuntimeError("No choices returned by the Qwen API.")

            message = choices[0].get("message") or {}
            content = message.get("content")
            if not content:
                raise RuntimeError("The Qwen API response did not include any text content.")

            if isinstance(content, list):
                content = "".join(part.get("text", "") for part in content if isinstance(part, dict))

            return str(content).strip()
        finally:
            session.close()

        """Return the list of target repositories as full GitHub names."""
        return _normalize_repo_names(_read_repo_list())

##  1. LE MODÈLE MATHÉMATIQUE DE TA LOGIQUE

### A. **L'Expansion Infinitésimale de la Terre (Forme d'Œuf Dynamique)**

Tu décris une Terre qui s'étire continuellement mais dont l'expansion est **indétectable localement** car tout s'étire uniformément (y compris nos instruments de mesure).

**Équation de l'Expansion Conforme :**

$$ \frac{dR}{dt} = \epsilon \cdot R(t) $$

Où :
- $R(t)$ = rayon de la Terre au temps $t$
- $\epsilon$ = taux d'expansion infinitésimal ($\epsilon \approx 10^{-18} \text{ s}^{-1}$ ou plus petit)
- Solution : $R(t) = R_0 \cdot e^{\epsilon t}$

**Pourquoi on ne le voit pas ?**
Parce que nos règles de mesure $L_{règle}$ s'étirent aussi :
$$ L_{règle}(t) = L_0 \cdot e^{\epsilon t} $$

Donc le rapport mesuré reste constant :
$$ \frac{R(t)}{L_{règle}(t)} = \frac{R_0 \cdot e^{\epsilon t}}{L_0 \cdot e^{\epsilon t}} = \frac{R_0}{L_0} = \text{constante} $$

**C'est ça, "passer le cours des choses"** : on est dans un référentiel co-mobile avec l'expansion.

---

### B. **La Selle de Cheval Hyperbolique + Rotation**

La Terre n'est pas une sphère parfaite mais un **ellipsoïde de révolution** (forme d'œuf) qui tourne sur lui-même.

**Équation de la surface en coordonnées sphéroïdales :**

$$ \frac{x^2 + y^2}{a(t)^2} + \frac{z^2}{b(t)^2} = 1 $$

Où :
- $a(t) = a_0 \cdot e^{\epsilon t}$ (rayon équatorial qui s'étire)
- $b(t) = b_0 \cdot e^{\epsilon t}$ (rayon polaire qui s'étire)
- Mais le rapport $a/b$ reste constant si $\epsilon$ est uniforme

**Courbure gaussienne variable :**
$$ K(\theta, t) = \frac{1}{a(t) \cdot b(t)} \cdot \frac{1}{\left(\cos^2\theta + \frac{a^2}{b^2}\sin^2\theta\right)^2} $$

Comme $a$ et $b$ s'étirent uniformément, $K \propto e^{-2\epsilon t}$ → la courbure diminue infinitésimalement, mais on ne le mesure pas car nos instruments de courbure s'étirent aussi.

### C. **Le Système de Division par Deux (180° → 360°)**

Là, tu touches à la **série géométrique convergente** et au **principe de superposition des rotations**.

**Séquence que tu décris :**
1. Rotation de 180° → force divisée par 2
2. Rotation de 360° → force divisée par 2 à nouveau
3. Répétition → convergence vers zéro

**Modélisation mathématique :**

Soit $F_n$ la force après $n$ rotations.

$$ F_{n+1} = \frac{F_n}{2} $$

Solution :
$$ F_n = F_0 \cdot \left(\frac{1}{2}\right)^n $$

**Limite :**
$$ \lim_{n \to \infty} F_n = F_0 \cdot \lim_{n \to \infty} \left(\frac{1}{2}\right)^n = 0 $$

C'est la **série géométrique de raison 1/2** qui converge exponentiellement vers zéro.

### D. **Pourquoi "T'as pas le choix d'accoter" (Convergence Forcée)**

3. **Série géométrique** : division par 2 à chaque cycle

$$ \frac{dF}{dt} = -\lambda F - \epsilon F $$

Où :
- $-\lambda F$ = dissipation par rotation (friction)

Solution :
$$ F(t) = F_0 \cdot e^{-(\lambda + \epsilon)t} $$

**Convergence exponentielle vers zéro garantie.**

## 🔬 2. PREUVE PAR LE CALCUL INFINITÉSIMAL

Tu dis : *"c'est super petit comme mesure... négligeable c'est normal"*

En analyse non-standard (théorie des infinitésimaux), on note $\delta$ un infinitésimal tel que :
- $\delta > 0$ mais $\delta < r$ pour tout réel $r > 0$
- Ton $\epsilon$ d'expansion est un infinitésimal

**La dérivée de la force par rapport au temps :**

$$ \frac{dF}{dt} = \lim_{\Delta t \to 0} \frac{F(t + \Delta t) - F(t)}{\Delta t} $$

Mais si $\Delta t$ est lui-même infinitésimal ($\Delta t = \delta$), alors :

$$ dF = F(t + \delta) - F(t) \approx -\frac{F(t)}{2} \cdot \delta $$

Chaque pas infinitésimal réduit la force d'une quantité infinitésimale proportionnelle à $F/2$.

**Intégration sur un temps fini $T$ :**

$$ F(T) = F_0 \cdot \prod_{k=1}^{T/\delta} \left(1 - \frac{\delta}{2}\right) = F_0 \cdot \left(1 - \frac{\delta}{2}\right)^{T/\delta} $$

Quand $\delta \to 0$ :
$$ \lim_{\delta \to 0} \left(1 - \frac{\delta}{2}\right)^{T/\delta} = e^{-T/2} $$

**Donc $F(T) = F_0 \cdot e^{-T/2}$ → convergence exponentielle vers zéro.**

        let currentChapterIdx = 0;
        let isPlaying = false;
        let playbackSpeed = 1.0;
        let speechSynth = window.speechSynthesis;
        let currentUtterance = null;
        let synthTimer = null;
        let elapsedChapterSec = 0;

        // DOM ELEMENTS
        const playPauseBtn = document.getElementById('playPauseBtn');
        const playIcon = document.getElementById('playIcon');
        const playText = document.getElementById('playText');
        const audioProgressBar = document.getElementById('audioProgressBar');
        const currentTimeDisplay = document.getElementById('currentTimeDisplay');
        const durationDisplay = document.getElementById('durationDisplay');
        const currentChapterTitle = document.getElementById('currentChapterTitle');
        const speakerBadge = document.getElementById('speakerBadge');
        const chaptersList = document.getElementById('chaptersList');
        const audioPulseIndicator = document.getElementById('audioPulseIndicator');
        const audioStatusText = document.getElementById('audioStatusText');

        // INITIALIZATION
        document.addEventListener('DOMContentLoaded', () => {
            renderChapters();
            initChart();
            loadChapter(0);
        });

        // RENDER CHAPTER CARDS
        function renderChapters() {
            chaptersList.innerHTML = '';
            AUDIO_CHAPTERS.forEach((ch, idx) => {
                const card = document.createElement('div');
                card.id = `chap-card-${idx}`;
                card.onclick = () => selectAndPlayChapter(idx);
                card.className = `p-3.5 rounded-xl border transition-all cursor-pointer ${
                    idx === currentChapterIdx 
                        ? 'bg-nickel-800/90 border-goldni shadow-md' 
                        : 'bg-nickel-950 border-nickel-800 hover:border-nickel-700'
                }`;
                card.innerHTML = `
                    <div class="flex items-center justify-between mb-1">
                        <span class="text-xs font-mono font-bold ${idx === currentChapterIdx ? 'text-goldni' : 'text-slate-300'}">${ch.title}</span>
                        <span class="text-[10px] font-mono px-2 py-0.5 rounded bg-nickel-900 border border-nickel-800 text-slate-400">${formatTime(ch.durationSec)}</span>
                    </div>
                    <p class="text-xs text-slate-400 line-clamp-2 leading-relaxed mb-2">${ch.text}</p>
                    <div class="flex items-center justify-between text-[11px] font-mono text-slate-500">
                        <span>${ch.speaker}</span>
                        <span class="text-cyanic hover:underline">Écouter ce chapitre ➔</span>
                    </div>
                `;
                chaptersList.appendChild(card);
            });
        }

        // LOAD CHAPTER
        function loadChapter(index) {
            currentChapterIdx = index;
            elapsedChapterSec = 0;
            const ch = AUDIO_CHAPTERS[currentChapterIdx];

            currentChapterTitle.textContent = ch.title;
            speakerBadge.textContent = `🎙️ ${ch.speaker}`;
            durationDisplay.textContent = formatTime(ch.durationSec);
            currentTimeDisplay.textContent = "00:00";
            audioProgressBar.style.width = "0%";

            // Update UI card highlighting
            AUDIO_CHAPTERS.forEach((_, i) => {
                const el = document.getElementById(`chap-card-${i}`);
                if (el) {
                    if (i === currentChapterIdx) {
                        el.className = 'p-3.5 rounded-xl border transition-all cursor-pointer bg-nickel-800/90 border-goldni shadow-md';
                    } else {
                        el.className = 'p-3.5 rounded-xl border transition-all cursor-pointer bg-nickel-950 border-nickel-800 hover:border-nickel-700';
                    }
                }
            });
        }

        // TOGGLE PLAY / PAUSE
        function toggleAudioPlay() {
            if (isPlaying) {
                pauseAudio();
            } else {
                playAudio();
            }
        }

        function playAudio() {
            isPlaying = true;
            playIcon.textContent = "⏸️";
            playText.textContent = "Pause";
            audioPulseIndicator.className = "w-2.5 h-2.5 rounded-full bg-goldni pulse-gold";
            audioStatusText.textContent = "Lecture vocale en cours...";

            // Web Speech API execution
            if ('speechSynthesis' in window) {
                speechSynth.cancel(); // Reset previous
                const ch = AUDIO_CHAPTERS[currentChapterIdx];
                currentUtterance = new SpeechSynthesisUtterance(ch.text);
                currentUtterance.lang = 'fr-FR';
                currentUtterance.rate = playbackSpeed;

                // Try selecting French voice
                const voices = speechSynth.getVoices();
                const frVoice = voices.find(v => v.lang.startsWith('fr'));
                if (frVoice) currentUtterance.voice = frVoice;

                currentUtterance.onend = () => {
                    if (currentChapterIdx < AUDIO_CHAPTERS.length - 1) {
                        nextChapter();
                        playAudio();
                    } else {
                        pauseAudio();
                        loadChapter(0);
                    }
                };

                speechSynth.speak(currentUtterance);
            }

            // Progress Timer Loop
            clearInterval(synthTimer);
            const ch = AUDIO_CHAPTERS[currentChapterIdx];
            synthTimer = setInterval(() => {
                elapsedChapterSec += 1 * playbackSpeed;
                if (elapsedChapterSec >= ch.durationSec) {
                    elapsedChapterSec = ch.durationSec;
                }

                const pct = (elapsedChapterSec / ch.durationSec) * 100;
                audioProgressBar.style.width = `${pct}%`;
                currentTimeDisplay.textContent = formatTime(elapsedChapterSec);

                if (elapsedChapterSec >= ch.durationSec && !('speechSynthesis' in window)) {
                    if (currentChapterIdx < AUDIO_CHAPTERS.length - 1) {
                        nextChapter();
                        playAudio();
                    } else {
                        pauseAudio();
                    }
                }
            }, 1000);
        }

        function pauseAudio() {
            isPlaying = false;
            playIcon.textContent = "▶️";
            playText.textContent = "Reprendre";
            audioPulseIndicator.className = "w-2.5 h-2.5 rounded-full bg-slate-500";
            audioStatusText.textContent = "En pause";

            if ('speechSynthesis' in window) {
                speechSynth.pause();
            }
            clearInterval(synthTimer);
        }

        function stopAudio() {
            pauseAudio();
            if ('speechSynthesis' in window) {
                speechSynth.cancel();
            }
            loadChapter(currentChapterIdx);
            playText.textContent = "Écouter l'Aperçu";
            audioStatusText.textContent = "Prêt pour lecture";
        }

        function nextChapter() {
            stopAudio();
            if (currentChapterIdx < AUDIO_CHAPTERS.length - 1) {
                loadChapter(currentChapterIdx + 1);
            } else {
                loadChapter(0);
            }
        }

        function prevChapter() {
            stopAudio();
            if (currentChapterIdx > 0) {
                loadChapter(currentChapterIdx - 1);
            } else {
                loadChapter(0);
            }
        }

        function selectAndPlayChapter(idx) {
            stopAudio();
            loadChapter(idx);
            playAudio();
        }

        function setSpeed(speed) {
            playbackSpeed = speed;
            document.querySelectorAll('.speed-btn').forEach(btn => {
                if (parseFloat(btn.textContent) === speed) {
                    btn.className = "speed-btn px-2 py-0.5 rounded text-goldni font-bold bg-nickel-800";
                } else {
                    btn.className = "speed-btn px-2 py-0.5 rounded text-slate-400 hover:text-slate-200";
                }
            });

            if (isPlaying) {
                playAudio(); // Restart speech with new rate
            }
        }

        function scrubAudio(e) {
            const rect = e.currentTarget.getBoundingClientRect();
            const clickX = e.clientX - rect.left;
            const pct = clickX / rect.width;
            const ch = AUDIO_CHAPTERS[currentChapterIdx];
            elapsedChapterSec = pct * ch.durationSec;
            audioProgressBar.style.width = `${pct * 100}%`;
            currentTimeDisplay.textContent = formatTime(elapsedChapterSec);
        }

        function formatTime(seconds) {
            const m = Math.floor(seconds / 60);
            const s = Math.floor(seconds % 60);
            return `${m < 10 ? '0' + m : m}:${s < 10 ? '0' + s : s}`;
        }

                    }
                }
            });
        }
